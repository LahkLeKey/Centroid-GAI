cmake_minimum_required(VERSION 3.20)
project(centroid_installed_consumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CENTROID_TEST_TRAINING)
  find_package(Centroid 0.1.0 EXACT CONFIG REQUIRED COMPONENTS Runtime Training)
else()
  find_package(Centroid 0.1.0 EXACT CONFIG REQUIRED COMPONENTS Runtime)
endif()
if(CENTROID_TEST_PREFIX)
  get_target_property(_centroid_includes Centroid::Runtime INTERFACE_INCLUDE_DIRECTORIES)
  foreach(_centroid_include IN LISTS _centroid_includes)
    string(FIND "${_centroid_include}" "${CENTROID_TEST_PREFIX}/" _centroid_prefix_at)
    if(NOT _centroid_prefix_at EQUAL 0)
      message(FATAL_ERROR "Installed include escapes the relocated prefix: ${_centroid_include}")
    endif()
  endforeach()
  get_target_property(_centroid_configurations Centroid::Runtime IMPORTED_CONFIGURATIONS)
  foreach(_centroid_configuration IN LISTS _centroid_configurations)
    get_target_property(_centroid_library Centroid::Runtime "IMPORTED_LOCATION_${_centroid_configuration}")
    string(FIND "${_centroid_library}" "${CENTROID_TEST_PREFIX}/" _centroid_prefix_at)
    if(NOT _centroid_prefix_at EQUAL 0)
      message(FATAL_ERROR "Installed library escapes the relocated prefix: ${_centroid_library}")
    endif()
  endforeach()
endif()
add_executable(centroid_consumer consumer.c)
target_link_libraries(centroid_consumer PRIVATE Centroid::Runtime)
add_executable(centroid_deployment_consumer deployment.c)
target_link_libraries(centroid_deployment_consumer PRIVATE Centroid::Runtime)
if(CENTROID_TEST_TRAINING)
  add_executable(centroid_training_consumer training.c)
  target_link_libraries(centroid_training_consumer PRIVATE Centroid::Training)
endif()
set(_centroid_consumers centroid_consumer centroid_deployment_consumer)
if(CENTROID_TEST_TRAINING)
  list(APPEND _centroid_consumers centroid_training_consumer)
endif()
foreach(_centroid_consumer IN LISTS _centroid_consumers)
  if(MSVC)
    target_compile_options(${_centroid_consumer} PRIVATE /W4 /WX)
  else()
    target_compile_options(${_centroid_consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
get_target_property(_centroid_type Centroid::Runtime TYPE)
if(WIN32 AND _centroid_type STREQUAL "SHARED_LIBRARY")
  add_custom_command(TARGET centroid_consumer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:Centroid::Runtime>" "$<TARGET_FILE_DIR:centroid_consumer>")
  add_custom_command(TARGET centroid_deployment_consumer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:Centroid::Runtime>" "$<TARGET_FILE_DIR:centroid_deployment_consumer>")
endif()
enable_testing()
add_test(NAME installed_consumer COMMAND centroid_consumer)
if(CENTROID_TEST_TRAINING)
  if(WIN32 AND _centroid_type STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET centroid_training_consumer POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:Centroid::Runtime>" "$<TARGET_FILE_DIR:centroid_training_consumer>")
  endif()
  add_test(NAME installed_training_consumer COMMAND centroid_training_consumer)
endif()
